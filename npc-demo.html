// 1. Include the script
<script src="npc-dialog.js"></script>

// 2. Create NPC dialog (after model is loaded)
const npc = new NPCDialog({
  engine: webllmEngine,
  npcName: "The Wise Wizard",
  npcPersonality: "an ancient wizard...",
  questGoal: "Player must answer riddles...",
  maxTurns: 5,
  onMissionComplete: () => {
    // Quest done! Add bonus points, etc.
    addBonusPoints(500);
  },
  onDialogClose: (success) => {
    resumeGame();
  }
});

// 3. At game start - get player name
npc.showNameInput((name) => {
  startGame();
});

// 4. When 4 lines are cleared:
if (linesCleared >= 4) {
  pauseGame();
  npc.startDialog((success) => {
    if (success) questsCompleted++;
  });
  linesCleared = 0;
}

// 5. Check total quests for win condition
if (questsCompleted >= 3) {
  showVictoryScreen();
}
