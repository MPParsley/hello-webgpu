Integration Code

// 1. Include the script
<script src="quiz-overlay.js"></script>

// 2. Create quiz instance
const quiz = new QuizOverlay({
  questions: YOUR_QUESTIONS,  // Optional custom questions
  onCorrect: () => addBonusPoints(100),
  onWrong: () => speedUpGame(),
  onComplete: (score, total, passed) => {
    if (passed) showVictoryScreen();
    else showGameOver();
  }
});

// 3. Show name input at game start
quiz.showNameInput((name) => {
  startGame();
});

// 4. In your game loop, when 4 lines are cleared:
if (linesCleared >= 4) {
  pauseGame();
  quiz.showQuestion((wasCorrect) => {
    resumeGame();
  });
  linesCleared = 0;
}

// 5. Check for mission complete
if (quiz.isComplete()) {
  if (quiz.isPassed()) {
    // Player won!
  }
}
